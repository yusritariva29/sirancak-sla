function authenticateUser(username, password) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Users');
  if (!sheet) return { success: false, message: "Database User belum disiapkan!" };
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] == username && data[i][1] == password) {
      return { success: true, username: data[i][0], role: data[i][2], nama: data[i][3], unit: data[i][4] };
    }
  }
  return { success: false, message: "Username atau Password salah!" };
}

// ================= FITUR FORM A & EARLY WARNING TEMUAN =================
function getFormA() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('FormA');
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  var list = [];
  
  for (var i = 1; i < data.length; i++) {
    var tglFormA = data[i][1];
    var adaDugaan = data[i][6]; // "Ada" atau "Tidak Ada"
    var sisaHari = 0;
    var kuadran = "Aman";
    var warningClass = "bg-emerald-100 text-emerald-800 border-emerald-300";

    // Jika ada dugaan, hitung SLA penetapan Temuan max 7 hari kerja sejak Form A dibuat
    if (tglFormA && adaDugaan === 'Ada') {
      sisaHari = hitungSisaHariKerjaCustom(new Date(tglFormA), 7);
      if (sisaHari >= 3) {
        kuadran = "Aman (Penetapan Temuan)";
        warningClass = "bg-emerald-100 text-emerald-800 border-emerald-300";
      } else if (sisaHari === 2) {
        kuadran = "Waspada Penetapan";
        warningClass = "bg-amber-100 text-amber-800 border-amber-300";
      } else if (sisaHari === 1) {
        kuadran = "Siaga Penetapan";
        warningClass = "bg-blue-100 text-blue-800 border-blue-300";
      } else {
        kuadran = "Bahaya / Batas 7 Hari Lewat";
        warningClass = "bg-rose-100 text-rose-800 border-rose-300 animate-pulse";
      }
    } else {
      kuadran = "Pengawasan Normal";
      warningClass = "bg-slate-100 text-slate-700 border-slate-300";
    }

    list.push({
      id: data[i][0],
      tanggal: tglFormA ? Utilities.formatDate(new Date(tglFormA), Session.getScriptTimeZone(), "yyyy-MM-dd") : "",
      no_form_a: data[i][2],
      pengawas: data[i][3],
      tahapan: data[i][4],
      uraian_pengawasan: data[i][5],
      ada_dugaan: adaDugaan,
      file_url: data[i][7] || "#",
      pembuat: data[i][8] || "-",
      sisa_hari: sisaHari,
      kuadran: kuadran,
      warningClass: warningClass
    });
  }
  return list;
}

function addFormA(form) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('FormA');
  if(!sheet) return "Error: Sheet 'FormA' tidak ditemukan!";

  return "Form A berhasil diregistrasi!";
}

// ================= FITUR KASUS, KAJIAN AWAL (2 HARI) & SLA LAPORAN (7 HARI) =================
function getCases() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Kasus');
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  var cases = [];
  for (var i = 1; i < data.length; i++) {
    var tglMasuk = data[i][1];
    var status = data[i][15] || 'Proses';
    var sisaHari = 0;
    var kuadran = "Aman";
    var warningClass = "bg-emerald-100 text-emerald-800 border-emerald-300";

    if (tglMasuk && status === 'Proses') {
      // SLA Penanganan Laporan total 7 hari kerja
      sisaHari = hitungSisaHariKerjaCustom(new Date(tglMasuk), 7);
      
      // Early Warning Kajian Awal (Paling lama 2 hari kerja)
      var sisaHariKajian = hitungSisaHariKerjaCustom(new Date(tglMasuk), 2);

      if (sisaHariKajian <= 0 && sisaHari > 0) {
        kuadran = "Perhatian: Deadline Kajian Awal (2 Hari)";
        warningClass = "bg-amber-100 text-amber-800 border-amber-300";
      } else if (sisaHari >= 4) {
        kuadran = "Aman (Penyusunan Kajian)";
        warningClass = "bg-emerald-100 text-emerald-800 border-emerald-300";
      } else if (sisaHari === 3) {
        kuadran = "Siaga Penanganan";
        warningClass = "bg-blue-100 text-blue-800 border-blue-300";
      } else if (sisaHari === 2) {
        kuadran = "Waspada Batas Waktu";
        warningClass = "bg-amber-100 text-amber-800 border-amber-300";
      } else {
        kuadran = "Bahaya / SLA 7 Hari Lewat";
        warningClass = "bg-rose-100 text-rose-800 border-rose-300 animate-pulse";
      }
    } else if (status === 'Selesai') {
      kuadran = "Selesai";
      warningClass = "bg-slate-100 text-slate-700 border-slate-300";
    }

    cases.push({
      id: data[i][0],
      tanggal_masuk: tglMasuk ? Utilities.formatDate(new Date(tglMasuk), Session.getScriptTimeZone(), "yyyy-MM-dd") : "",
      sumber: data[i][2],
      referensi_no: data[i][3] || "-",
      nama_pelapor: data[i][3],
      nama_terlapor: data[i][4],
      peristiwa: data[i][5],
      tempat_kejadian: data[i][6],
      tgl_kejadian: data[i][7],
      tgl_diketahui: data[i][8],
      saksi: data[i][9],
      bukti: data[i][10],
      uraian: data[i][11],
      penerima_laporan: data[i][12] || "Muhammad Ramadhan",
      status: status,
      tindak_lanjut: data[i][16],
      sisa_hari: sisaHari,
      kuadran: kuadran,
      warningClass: warningClass
    });
  }
  return cases;
}

// Helper Penghitung Hari Kerja Kustom (Lewati Sabtu & Minggu)
function hitungSisaHariKerjaCustom(startDate, maxDays) {
  var today = new Date();
  today.setHours(0,0,0,0);
  var targetDate = new Date(startDate);
  targetDate.setHours(0,0,0,0);
  
  var hariKerjaLewat = 0;
  var cursor = new Date(targetDate);
  while(hariKerjaLewat < maxDays) {
    cursor.setDate(cursor.getDate() + 1);
    var day = cursor.getDay();
    if (day !== 0 && day !== 6) hariKerjaLewat++;
  }
  
  var sisaHariKerja = 0;
  var tempToday = new Date(today);
  while(tempToday < cursor) {
    tempToday.setDate(tempToday.getDate() + 1);
    var day = tempToday.getDay();
    if (day !== 0 && day !== 6) sisaHariKerja++;
  }
  if (today > cursor) return 0;
  return sisaHariKerja;
}
